# V0.2 — Events & DX Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Establish the `lemon-widgets` crate boundary, CI/license baseline, two new examples, and full keyboard + hover event infrastructure.

**Architecture:** Extract widget builders to `crates/lemon-widgets/` (re-exporting from `lemon`). Add `KeyEvent`, `Cursor`, `focusable` types to the element layer. Extend `EventHandlers` and `BoxElement` with hover/keyboard handlers. Add `FocusManager` and hover-tracking to `AppState`.

**Tech Stack:** Rust, winit 0.30, Taffy 0.7, Vello 0.9, GitHub Actions.

```rust
use lemon::{run, Color, Cx, WindowConfig};
use lemon_widgets::{Button, Column, Component, Row, Text};

#[derive(Clone)]
struct Item {
    id: u64,
    label: String,
}

fn app(cx: &Cx) -> lemon::element::Element {
    let items = cx.use_signal(Vec::<Item>::new());
    let next_id = cx.use_signal(0u64);

    let items_read = items.clone();
    let items_add = items.clone();
    let next_id_add = next_id.clone();

    Column::new()
        .padding(24.0)
        .gap(12.0)
        .child(Text::new("Keyed List").font_size(20.0))
        .child(
            Button::new("Add item").on_click(move || {
                let id = next_id_add.get();
                next_id_add.update(|n| *n += 1);
                items_add.update(|list| {
                    list.push(Item {
                        id,
                        label: format!("Item #{id}"),
                    });
                });
            }),
        )
        .child({
            let items_snapshot = items_read.get();
            let mut col = Column::new().gap(8.0);
            for item in items_snapshot {
                let items_remove = items.clone();
                let item_id = item.id;
                let label = item.label.clone();
                col = col.child(
                    Component::new(move |_cx: &Cx| {
                        let items_rm = items_remove.clone();
                        let id = item_id;
                        Row::new()
                            .gap(8.0)
                            .align_items(lemon::element::style::Align::Center)
                            .child(
                                Text::new(label.clone())
                                    .font_size(15.0)
                                    .color(Color::rgb8(210, 210, 230)),
                            )
                            .child(
                                Button::new("Remove")
                                    .background(Color::rgb8(180, 50, 50))
                                    .on_click(move || {
                                        items_rm.update(|list| list.retain(|i| i.id != id));
                                    }),
                            )
                            .into_element()
                    })
                    .key(item.id),
                );
            }
            col
        })
        .into_element()
}

fn main() {
    run(
        WindowConfig::default()
            .title("Keyed List")
            .size(500.0, 600.0),
        app,
    );
}
```

use taffy::NodeId;
use crate::retained::{RetainedNode, RetainedTree};

pub struct FocusManager {
    pub focused: Option<NodeId>,
}

impl FocusManager {
    pub fn new() -> Self {
        Self { focused: None }
    }

    /// Advance focus to the next/previous focusable node in document order.
    pub fn cycle(&mut self, tree: &RetainedTree, forward: bool) {
        let focusable = collect_focusable(tree);
        if focusable.is_empty() {
            return;
        }
        let current = self.focused.and_then(|id| focusable.iter().position(|&f| f == id));
        let len = focusable.len();
        let next = match current {
            None => 0,
            Some(i) if forward => (i + 1) % len,
            Some(i) => (i + len - 1) % len,
        };
        self.focused = Some(focusable[next]);
    }

    /// Set focus to the node with the given taffy id (if it is focusable).
    pub fn focus_by_id(&mut self, id: NodeId, tree: &RetainedTree) {
        let focusable = collect_focusable(tree);
        if focusable.contains(&id) {
            self.focused = Some(id);
        }
    }
}

impl Default for FocusManager {
    fn default() -> Self {
        Self::new()
    }
}

/// Walk the retained tree in pre-order and collect taffy IDs of focusable nodes.
pub fn collect_focusable(tree: &RetainedTree) -> Vec<NodeId> {
    let mut out = Vec::new();
    if let Some(root) = &tree.root {
        collect_focusable_node(root, &mut out);
    }
    out
}

fn collect_focusable_node(node: &RetainedNode, out: &mut Vec<NodeId>) {
    if node.style.focusable {
        if let Some(id) = node.taffy_id {
            out.push(id);
        }
    }
    for child in &node.children {
        collect_focusable_node(child, out);
    }
}

/// Walk the retained tree in post-order and return the innermost node under `point`
/// that has a hover handler or a non-default cursor.
pub fn hit_test_hover<'a>(
    node: &'a RetainedNode,
    layout: &LayoutMap,
    point: LogicalPoint,
) -> Option<&'a RetainedNode> {
    if matches!(node.kind, RetainedKind::Component { .. }) {
        let mut hit = None;
        for child in &node.children {
            hit = hit_test_hover(child, layout, point).or(hit);
        }
        return hit;
    }

    let mut hit = None;
    for child in &node.children {
        hit = hit_test_hover(child, layout, point).or(hit);
    }

    let is_hoverable = node.handlers.on_hover_enter.is_some()
        || node.handlers.on_hover_leave.is_some()
        || node.style.cursor != crate::element::events::Cursor::Default;

    if is_hoverable {
        if let Some(id) = node.taffy_id {
            if let Some(rect) = layout.get(id) {
                if point_in_rect(point, rect) {
                    hit = Some(node);
                }
            }
        }
    }

    hit
}

    }
    for child in &node.children {
        if let Some(found) = find_node_by_taffy_id(child, id) {
            return Some(found);
        }
    }

}

fn event_pass_keyboard(&mut self, key_event: winit::event::KeyEvent) {
    let lemon_key = winit_key_to_lemon(&key_event.logical_key);
    let state = if key_event.state == ElementState::Pressed {
        KeyState::Pressed
    } else {
        KeyState::Released
    };

    // Tab: cycle focus (consume before dispatching to focused node)
    if state == KeyState::Pressed
        && !key_event.repeat
        && lemon_key == LemonKey::Named(NamedKey::Tab)
    {
        if let Some(tree) = self.retained.as_ref() {
            self.focus_manager.cycle(tree, !self.active_modifiers.shift);
        }
        self.paint_dirty = true;
        return;
    }

    let Some(focused_id) = self.focus_manager.focused else {
        return;
    };
    let Some(tree) = self.retained.as_ref() else {
        return;
    };
    let Some(root) = tree.root.as_ref() else {
        return;
    };
    let Some(node) = find_node_by_taffy_id(root, focused_id) else {
        return;
    };

    let ev = KeyEvent {
        key: lemon_key,
        modifiers: self.active_modifiers.clone(),
        repeat: key_event.repeat,
        state,
    };

    match ev.state {
        KeyState::Pressed => {
            if let Some(f) = &node.handlers.on_key_down {
                f(ev);
                self.layout_dirty = true;
                self.paint_dirty = true;
            }
        }
        KeyState::Released => {
            if let Some(f) = &node.handlers.on_key_up {
                f(ev);
                self.layout_dirty = true;
                self.paint_dirty = true;
            }
        }
    }
}

fn event_pass_hover(&mut self, point: LogicalPoint) {
    let new_id = self
        .retained
        .as_ref()
        .and_then(|t| t.root.as_ref())
        .and_then(|root| hit_test_hover(root, &self.layout_map, point))
        .and_then(|node| node.taffy_id);

    if new_id == self.hovered_node_id {
        return;
    }

    // Dispatch leave on old hovered node
    if let Some(old_id) = self.hovered_node_id {
        if let Some(root) = self.retained.as_ref().and_then(|t| t.root.as_ref()) {
            if let Some(old_node) = find_node_by_taffy_id(root, old_id) {
                if let Some(f) = &old_node.handlers.on_hover_leave {
                    f();
                }
            }
        }
    }

    // Dispatch enter on new hovered node + apply cursor
    let new_cursor = if let Some(new_id) = new_id {
        if let Some(root) = self.retained.as_ref().and_then(|t| t.root.as_ref()) {
            if let Some(new_node) = find_node_by_taffy_id(root, new_id) {
                if let Some(f) = &new_node.handlers.on_hover_enter {
                    f();
                }
                new_node.style.cursor.clone()
            } else {
                Cursor::Default
            }
        } else {
            Cursor::Default
        }
    } else {
        Cursor::Default
    };

    if let Some(window) = &self.window {
        window.set_cursor(lemon_cursor_to_winit(&new_cursor));
    }

    self.hovered_node_id = new_id;
    self.layout_dirty = true;
    self.paint_dirty = true;
}
